# 🎨 头像框商城系统设计方案

## 📋 一、系统概述

将 `xiangkuang` 目录下的 880+ 张头像框图片作为商城商品，支持用户购买和装备。GIF动态头像框定价更高，VIP用户享受折扣优惠。

## 🎯 二、核心功能设计

### 2.1 头像框规格标准化

**固定尺寸规范：**
- **头像框外框尺寸**: 200dp × 200dp（固定）
- **内部头像区域**: 160dp × 160dp（圆形裁剪）
- **边框宽度**: 20dp
- **头像框类型**:
  - PNG静态框：标准商品
  - GIF动态框：高级商品（价格2-3倍）

**适配方案：**
```kotlin
// 头像框组件 - 固定尺寸，完美适配
@Composable
fun AvatarWithFrame(
    avatarUri: String?,
    frameAssetPath: String?,
    size: Dp = 200.dp,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier.size(size),
        contentAlignment = Alignment.Center
    ) {
        // 1. 底层：用户头像（圆形裁剪，160dp）
        AsyncImage(
            model = avatarUri,
            contentDescription = "用户头像",
            modifier = Modifier
                .size(160.dp)
                .clip(CircleShape),
            contentScale = ContentScale.Crop
        )
        
        // 2. 顶层：头像框（200dp，覆盖整个区域）
        if (frameAssetPath != null) {
            if (frameAssetPath.endsWith(".gif")) {
                // GIF动态框
                GifImage(
                    assetPath = frameAssetPath,
                    modifier = Modifier.fillMaxSize()
                )
            } else {
                // PNG静态框
                Image(
                    painter = painterResource(assetPath = frameAssetPath),
                    contentDescription = "头像框",
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Fit
                )
            }
        }
    }
}
```

```kotlin
fun determineRarity(folderName: String, fileName: String): ItemRarity {
    return when {
        // GIF动态框 - 高稀有度
        fileName.endsWith(".gif") -> {
            when {
                folderName.contains("8") -> ItemRarity.LEGENDARY  // 头像框8
                folderName.contains("6") -> ItemRarity.EPIC       // 头像框6
                else -> ItemRarity.RARE
            }
        }
        // PNG静态框 - 根据文件夹分配
        folderName.contains("A4") || folderName.contains("5") -> ItemRarity.COMMON
        folderName.contains("1") || folderName.contains("2") -> ItemRarity.COMMON
        folderName.contains("4") || folderName.contains("7") -> ItemRarity.RARE
        folderName.contains("6") || folderName.contains("8") -> ItemRarity.EPIC
        else -> ItemRarity.COMMON
    }
}
```

## 📦 五、数据初始化方案

### 5.1 自动扫描并导入头像框

```kotlin
class AvatarFrameInitializer(private val context: Context) {
    
    suspend fun initializeAvatarFrames(shopDao: ShopDao) {
        val assetManager = context.assets
        val baseFolder = "xiangkuang"
        
        // 扫描所有子文件夹
        val folders = assetManager.list(baseFolder) ?: return
        
        var sortOrder = 1
        
        folders.forEach { folderName ->
            val folderPath = "$baseFolder/$folderName"
            val files = assetManager.list(folderPath) ?: emptyArray()
            
            files.filter { it.endsWith(".png") || it.endsWith(".gif") }
                .forEach { fileName ->
                    val assetPath = "$folderPath/$fileName"
                    val isAnimated = fileName.endsWith(".gif")
                    val rarity = determineRarity(folderName, fileName)
                    val (price, vipPrice) = calculatePrice(rarity, isAnimated)
                    
                    val shopItem = ShopItem(
                        name = generateFrameName(folderName, fileName),
                        description = generateDescription(rarity, isAnimated),
                        icon = if (isAnimated) "✨" else "🖼️",
                        price = price,
                        vipPrice = vipPrice,
                        type = "avatar_frame",
                        value = 1,
                        isAvailable = true,
                        assetPath = assetPath,
                        rarity = rarity.name,
                        isAnimated = isAnimated,
                        category = folderName,
                        sortOrder = sortOrder++
                    )
                    
                    shopDao.insertShopItem(shopItem)
                }
        }
        
        Log.d("AvatarFrameInit", "Initialized $sortOrder avatar frames")
    }
    
    private fun calculatePrice(rarity: ItemRarity, isAnimated: Boolean): Pair<Int, Int> {
        val basePrice = when (rarity) {
            ItemRarity.COMMON -> if (isAnimated) 300 else 50
            ItemRarity.RARE -> if (isAnimated) 500 else 100
            ItemRarity.EPIC -> if (isAnimated) 800 else 200
            ItemRarity.LEGENDARY -> if (isAnimated) 1200 else 300
        }
        val vipPrice = (basePrice * 0.8).toInt()  // VIP 8折
        return Pair(basePrice, vipPrice)
    }
    
    private fun generateFrameName(folder: String, file: String): String {
        // 从文件名提取编号
        val number = file.replace(Regex("[^0-9]"), "")
        return when {
            folder.contains("A4") -> "A4头像框 #$number"
            folder.contains("1") -> "紫宸系列 #$number"
            folder.contains("2") -> "紫宸雅韵 #$number"
            folder.contains("4") -> "紫宸华章 #$number"
            folder.contains("5") -> "经典头像框 #$number"
            folder.contains("6") -> "紫宸璀璨 #$number"
            folder.contains("7") -> "紫宸流光 #$number"
            folder.contains("8") -> "紫宸传说 #$number"
            else -> "头像框 #$number"
        }
    }
    
    private fun generateDescription(rarity: ItemRarity, isAnimated: Boolean): String {
        val rarityDesc = when (rarity) {
            ItemRarity.COMMON -> "简约大方"
            ItemRarity.RARE -> "精致优雅"
            ItemRarity.EPIC -> "华丽璀璨"
            ItemRarity.LEGENDARY -> "传说品质"
        }
        val animDesc = if (isAnimated) "，动态效果" else ""
        return "$rarityDesc$animDesc，彰显独特品味"
    }
}
```

```kotlin
class ShopViewModel(application: Application) : AndroidViewModel(application) {
    
    // ... 现有代码 ...
    
    // 🔥 新增：头像框相关
    val avatarFrames: StateFlow<List<ShopItem>> = shopRepository.getAvatarFrames()
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())
    
    val userOwnedFrames: StateFlow<List<UserAvatarFrame>> = 
        shopRepository.getUserOwnedFrames(getCurrentUserId())
            .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())
    
    val isUserVip: StateFlow<Boolean> = userRepository.isUserVip(getCurrentUserId())
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), false)
    
    // 购买头像框
    fun purchaseAvatarFrame(frame: ShopItem) {
        viewModelScope.launch {
            val userId = getCurrentUserId()
            val isVip = isUserVip.value
            val price = if (isVip) frame.vipPrice else frame.price
            
            // 检查是否已拥有
            val owned = shopRepository.isFrameOwned(userId, frame.id)
            if (owned) {
                _message.value = "您已拥有此头像框"
                return@launch
            }
            
            // 扣除金币
            val success = coinRepository.spendCoins(userId, price)
            if (success) {
                // 添加到用户拥有列表
                shopRepository.addUserFrame(userId, frame.id)
                
                // 记录购买历史
                val purchase = PurchaseHistory(
                    userId = userId,
                    itemId = frame.id,
                    itemName = frame.name,
                    price = price,
                    timestamp = LocalDateTime.now().toString()
                )
                shopRepository.insertPurchaseHistory(purchase)
                
                _message.value = "购买成功！"
            } else {
                _message.value = "金币不足"
            }
        }
    }
    
    // 装备头像框
    fun equipAvatarFrame(frameId: Int) {
        viewModelScope.launch {
            val userId = getCurrentUserId()
            userRepository.updateEquippedFrame(userId, frameId)
            _message.value = "装备成功！"
        }
    }
    
    // 卸下头像框
    fun unequipAvatarFrame() {
        viewModelScope.launch {
            val userId = getCurrentUserId()
            userRepository.updateEquippedFrame(userId, null)
            _message.value = "已卸下头像框"
        }
    }
}
```

### 8.1 VIP折扣机制

```kotlin
object VipBenefits {
    const val AVATAR_FRAME_DISCOUNT = 0.8  // 头像框8折
    const val SHOP_ITEM_DISCOUNT = 0.85    // 其他商品85折
    
    fun calculatePrice(originalPrice: Int, isVip: Boolean, itemType: String): Int {
        if (!isVip) return originalPrice
        
        val discount = when (itemType) {
            "avatar_frame" -> AVATAR_FRAME_DISCOUNT
            else -> SHOP_ITEM_DISCOUNT
        }
        
        return (originalPrice * discount).toInt()
    }
}
```

```kotlin
@Composable
fun GifImage(
    assetPath: String,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val imageLoader = ImageLoader.Builder(context)
        .components {
            add(ImageDecoderDecoder.Factory())
        }
        .build()
    
    AsyncImage(
        model = ImageRequest.Builder(context)
            .data("file:///android_asset/$assetPath")
            .build(),
        imageLoader = imageLoader,
        contentDescription = null,
        modifier = modifier,
        contentScale = ContentScale.Fit
    )
}
```
