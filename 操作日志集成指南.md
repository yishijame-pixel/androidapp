## 集成步骤

### 步骤1：更新 AppDatabase.kt

```kotlin
@Database(
    entities = [
        // ... 现有实体
        OperationLog::class  // 🔥 添加这一行
    ],
    version = 13,  // 🔥 版本号从12升级到13
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {
    
    // ... 现有 DAO
    
    abstract fun operationLogDao(): OperationLogDao  // 🔥 添加这一行
    
    companion object {
        // ... 现有代码
        
        // 🔥 添加迁移脚本
        private val MIGRATION_12_13 = object : Migration(12, 13) {
            override fun migrate(database: SupportSQLiteDatabase) {
                // 创建操作日志表
                database.execSQL("""
                    CREATE TABLE IF NOT EXISTS operation_logs (
                        id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                        userId INTEGER NOT NULL DEFAULT 0,
                        operation TEXT NOT NULL,
                        details TEXT NOT NULL,
                        result TEXT NOT NULL,
                        errorMessage TEXT NOT NULL DEFAULT '',
                        timestamp INTEGER NOT NULL
                    )
                """)
                
                // 创建索引以提高查询性能
                database.execSQL("""
                    CREATE INDEX IF NOT EXISTS index_operation_logs_userId 
                    ON operation_logs(userId)
                """)
                
                database.execSQL("""
                    CREATE INDEX IF NOT EXISTS index_operation_logs_operation 
                    ON operation_logs(operation)
                """)
                
                database.execSQL("""
                    CREATE INDEX IF NOT EXISTS index_operation_logs_timestamp 
                    ON operation_logs(timestamp)
                """)
            }
        }
        
        fun getDatabase(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(...)
                    .addMigrations(
                        // ... 现有迁移
                        MIGRATION_12_13  // 🔥 添加这一行
                    )
                    .build()
                INSTANCE = instance
                instance
            }
        }
    }
}
```

## 使用示例

### 查询最近的日志

```kotlin
class LogViewModel(application: Application) : AndroidViewModel(application) {
    private val logDao = AppDatabase.getDatabase(application).operationLogDao()
    
    val recentLogs: StateFlow<List<OperationLog>> = logDao.getRecentLogs(100)
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = emptyList()
        )
}
```

### 查询特定操作的日志

```kotlin
// 查询所有登录记录
val loginLogs = logDao.getLogsByOperation(OperationLog.OP_LOGIN, 50)

// 查询所有失败的操作
val failedLogs = logDao.getLogsByResult(OperationLog.RESULT_FAILED, 50)

// 查询特定时间范围的日志
val startTime = System.currentTimeMillis() - (7 * 24 * 60 * 60 * 1000L)  // 7天前
val endTime = System.currentTimeMillis()
val weekLogs = logDao.getLogsByDateRange(startTime, endTime)
```

## 日志清理策略

日志会自动清理，保留：
- 最近30天的日志
- 或最多10000条记录

可以在 `OperationLogDao.cleanOldLogs()` 中调整这些参数。

---

## 安全注意事项

1. **敏感信息**：不要在日志中记录密码、支付信息等敏感数据
2. **性能影响**：日志记录是异步的，不会阻塞主线程
3. **存储空间**：定期清理旧日志，避免占用过多空间
4. **隐私合规**：根据隐私政策，可能需要提供日志导出和删除功能

## 日志分析

可以基于日志数据进行分析：

```kotlin
// 统计用户活跃度
suspend fun getUserActivityStats(userId: Long): Map<String, Int> {
    val logs = logDao.getLogsByUser(userId, 1000).first()
    return logs.groupingBy { it.operation }.eachCount()
}

// 统计失败操作
suspend fun getFailureRate(): Float {
    val allLogs = logDao.getRecentLogs(1000).first()
    val failedCount = allLogs.count { it.result == OperationLog.RESULT_FAILED }
    return if (allLogs.isNotEmpty()) {
        failedCount.toFloat() / allLogs.size
    } else {
        0f
    }
}

// 统计金币流水
suspend fun getCoinFlowStats(): Pair<Int, Int> {
    val logs = logDao.getLogsByOperation(OperationLog.OP_SPIN, 1000).first()
    var totalCost = 0
    var totalReward = 0
    
    logs.forEach { log ->
        val json = JSONObject(log.details)
        totalCost += json.optInt("coinCost", 0)
        totalReward += json.optInt("coinReward", 0)
    }
    
    return Pair(totalCost, totalReward)
}
```
